cmake_minimum_required(VERSION 3.16)
project(sayitem_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${SAYITEM_JDK}/include/jni.h")
    message(FATAL_ERROR "Set SAYITEM_JDK to the JDK root (only JNI headers are needed)")
endif()
file(TO_CMAKE_PATH "${SAYITEM_JDK}" SAYITEM_JDK)
file(TO_CMAKE_PATH "${WHISPER_JNI_SOURCE}" WHISPER_JNI_SOURCE)
if(WIN32)
    set(JNI_PLATFORM win32)
elseif(APPLE)
    set(JNI_PLATFORM darwin)
else()
    set(JNI_PLATFORM linux)
endif()
set(JNI_INCLUDE_DIRS "${SAYITEM_JDK}/include" "${SAYITEM_JDK}/include/${JNI_PLATFORM}")
if(NOT WHISPER_JNI_SOURCE)
    message(FATAL_ERROR "Set WHISPER_JNI_SOURCE to the pinned JNI source checkout")
endif()
set(WHISPER_CPP_SOURCE "${WHISPER_JNI_SOURCE}/src/main/native/whisper")
include("${CMAKE_CURRENT_LIST_DIR}/patch.cmake")
add_subdirectory("${WHISPER_CPP_SOURCE}" whisper EXCLUDE_FROM_ALL)
add_library(sayitem-whisper SHARED
    "${CMAKE_CURRENT_LIST_DIR}/abi.cpp"
    "${WHISPER_JNI_SOURCE}/src/main/native/io_github_givimad_whisperjni_WhisperJNI.cpp"
    "${WHISPER_CPP_SOURCE}/examples/grammar-parser.cpp")
target_include_directories(sayitem-whisper PRIVATE
    ${JNI_INCLUDE_DIRS}
    "${WHISPER_JNI_SOURCE}/src/main/native"
    "${WHISPER_CPP_SOURCE}/include"
    "${WHISPER_CPP_SOURCE}/examples")
target_link_libraries(sayitem-whisper PRIVATE whisper ggml)
if(MSVC)
    target_sources(sayitem-whisper PRIVATE "${CMAKE_CURRENT_LIST_DIR}/exports.def")
    target_compile_options(sayitem-whisper PRIVATE /utf-8)
endif()
if(UNIX AND NOT APPLE)
    target_link_options(sayitem-whisper PRIVATE -static-libgcc -static-libstdc++)
endif()
set_target_properties(sayitem-whisper PROPERTIES PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/output" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/output"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/output" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/output")
add_executable(native-probe "${CMAKE_CURRENT_LIST_DIR}/probe.cpp")
target_link_libraries(native-probe PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(native-probe PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/output" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/output")
