cmake_minimum_required(VERSION 3.20)
project(sayitem_ct2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${CT2_DEPS}/dependencies.cmake")
if(WIN32)
    get_filename_component(TOOLS "${CMAKE_CXX_COMPILER}" DIRECTORY)
    execute_process(COMMAND "${TOOLS}/dumpbin.exe" /exports "${CT2_LIBRARY}" OUTPUT_VARIABLE EXPORTS COMMAND_ERROR_IS_FATAL ANY)
    string(REPLACE "\n" ";" LINES "${EXPORTS}")
    set(DEFINITION "LIBRARY ctranslate2\nEXPORTS\n")
    foreach(LINE IN LISTS LINES)
        if(LINE MATCHES "^[ \t]+[0-9]+[ \t]+[0-9A-F]+[ \t]+[0-9A-F]+[ \t]+([^ \r\n]+)")
            string(APPEND DEFINITION "${CMAKE_MATCH_1}\n")
        endif()
    endforeach()
    file(WRITE "${CMAKE_BINARY_DIR}/ctranslate2.def" "${DEFINITION}")
    set(LINK_LIBRARY "${CMAKE_BINARY_DIR}/ctranslate2.lib")
    execute_process(COMMAND "${CMAKE_AR}" "/def:${CMAKE_BINARY_DIR}/ctranslate2.def" /machine:X64 "/out:${LINK_LIBRARY}" COMMAND_ERROR_IS_FATAL ANY)
else()
    set(LINK_LIBRARY "${CT2_LIBRARY}")
endif()
add_library(sayitem-ct2 SHARED bridge.cpp)
target_include_directories(sayitem-ct2 PRIVATE "${CT2_SOURCE}/include")
target_link_libraries(sayitem-ct2 PRIVATE "${LINK_LIBRARY}")
if(MSVC)
    target_compile_options(sayitem-ct2 PRIVATE /utf-8 /d2FH4-)
endif()
if(APPLE)
    set(RPATH "@loader_path/ctranslate2/.dylibs;@loader_path/ctranslate2;@loader_path/.dylibs")
elseif(UNIX)
    set(RPATH "$ORIGIN/ctranslate2.libs;$ORIGIN/ctranslate2")
endif()
set_target_properties(sayitem-ct2 PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${RPATH}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/output" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/output"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/output" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/output")
